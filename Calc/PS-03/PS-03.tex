\begin{enumerate}
  \item $\lim_{t \to \infty} \frac{t^2}{t-t^2} = -1$
  \item $\lim_{x \to \infty} \frac{3x^2}{x^2-8x+15} = 3$
  \item $\lim_{x \to \infty} \frac{3\sqrt{x^3}+3x}{\sqrt{2x^3}} = \frac{3}{\sqrt{2}}$
  \item $\lim_{t \to \infty} \frac{\sqrt{2t+1}}{t+4} = 0$
\end{enumerate}

\section*{Solution 1}
\subsection*{Solution 1.1}
Let $\epsilon > 0$ and choose $M = 1+\frac{1}{\epsilon} > 1$. Then, if $t > M$, we have
\begin{align*}
    \left| f(t) - L \right|     
    &= \left| \frac{t^2}{t-t^2} - (-1) \right| \\
    &= \left| \frac{t^2}{t-t^2} + 1 \right| \\
    &= \left| \frac{t^2+(t-t^2)}{t-t^2} \right| \\
    &= \left| \frac{t}{t(1-t)} \right| && \text{since }t > M \geq 0 \\
    &= \frac{1}{\left|1-t\right|} \\
    &= \frac{1}{\left|t-1\right|} \\
    &= \frac{1}{t-1} && \text{since } t >M > 1 \\
    &< \frac{1}{M-1} && \text{since } \frac{1}{x-1} \text{ is decreasing for } x > 1 \\
    &= \frac{1}{(1+\frac{1}{\epsilon})-1} \\
    &= \epsilon \quad \qed
\end{align*}

\section*{Solution 2}
\subsection*{Solution 2.1}
Let $\epsilon > 0$ and choose $M = \max\left(16,\frac{48}{\epsilon}\right)$. Then, if $x > M$, we have 
\begin{align*}
    \left| f(x) - L \right|
    &=\left| \frac{3x^2}{x^2-8x+15} - 3 \right| \\
    &= \left| \frac{3x^2 - 3(x^2-8x+15)}{x^2-8x+15} \right| \\
    &= \left| \frac{24x - 45}{x^2-8x+15} \right| \\
    &= \frac{ \left|24x-45\right| }{ \left|x^2-8x+15\right| }
\end{align*}
Considering the numerator and denominator separately, we have
\[
    \left|24x-45\right| > 0 \quad \text{if }x > 2
\]
and
\[
    \left|x^2-8x+15\right| = \left|(x-3)(x-5)\right| > 0\quad \text{if } x > 5
\]
Let's assume, then, that $M \geq 5$. Therefore,
\begin{align*}
    \left|f(x)-L\right| = \ldots &= \frac{ \left|24x-45\right| }{ \left|x^2-8x+15\right| } \\
    &= \frac{24x-45}{x^2-8x+15} \\
    &< \frac{24x}{x^2-8x+15}
\end{align*}
Let's bound the denominator:
\[
    x^2-8x+15 > \frac{1}{2}x^2 \iff \frac{1}{2}x^2 - 8x + 15 > 0
\]
To find the zeros of the above quadratic expression, use the quadratic formula:
\[
    x = \frac{16 \pm \sqrt{156 - 4(30)}}{2} = \frac{16\pm \sqrt{244}}{2} = \frac{16 \pm 2\sqrt{61}}{2} = 8\pm \sqrt{61}
\]
Now, the larger root is $8 + \sqrt{61} < 16$. Therefore, for $x > 16$, we have
\[
    x^2-8x+15 > \frac{1}{2}x^2
\]
Therefore, we'll assume $M \geq 16$ and can say
\begin{align*}
    \left|f(x)-L\right| = \ldots &= \frac{24x-45}{x^2-8x+15} \\
    &< \frac{24x}{\frac{1}{2}x^2} \\ 
    &= \frac{48}{x} \\
    &= \frac{48}{48/\epsilon} \\
    &= \epsilon \quad \qed
\end{align*}
